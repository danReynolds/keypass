cmake_minimum_required(VERSION 3.22)
project(keypass_hardware LANGUAGES CXX)
if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Direct USB backend currently targets macOS and Linux")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(FIDO REQUIRED IMPORTED_TARGET libfido2>=1.16)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
set(DEPS "${CMAKE_CURRENT_BINARY_DIR}/deps")
file(MAKE_DIRECTORY "${DEPS}")
set(JSON_HASH "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
if(NOT EXISTS "${DEPS}/json.hpp")
  file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp"
    "${DEPS}/json.hpp" EXPECTED_HASH SHA256=${JSON_HASH} TLS_VERIFY ON)
endif()
file(SHA256 "${DEPS}/json.hpp" ACTUAL_JSON_HASH)
if(NOT ACTUAL_JSON_HASH STREQUAL JSON_HASH)
  message(FATAL_ERROR "JSON dependency checksum mismatch")
endif()
add_library(keypass_hardware SHARED keypass_hardware.cpp)
target_compile_features(keypass_hardware PRIVATE cxx_std_17)
target_compile_options(keypass_hardware PRIVATE -Wall -Wextra -Werror)
target_include_directories(keypass_hardware PRIVATE "${DEPS}")
target_link_libraries(keypass_hardware PRIVATE PkgConfig::FIDO OpenSSL::Crypto Threads::Threads)
set_target_properties(keypass_hardware PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
include(CTest)
if(BUILD_TESTING)
  add_executable(keypass_hardware_test test.cpp)
  target_compile_features(keypass_hardware_test PRIVATE cxx_std_17)
  target_compile_options(keypass_hardware_test PRIVATE -UNDEBUG)
  target_include_directories(keypass_hardware_test PRIVATE "${DEPS}")
  target_link_libraries(keypass_hardware_test PRIVATE keypass_hardware Threads::Threads)
  add_test(NAME hardware_native COMMAND keypass_hardware_test)
endif()
